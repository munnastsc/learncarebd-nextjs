var LCBD_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbzHMI1Ix_DpDj90ZxnTXlsYlzTq7H2-b9CX4a6fDa1s7ZxAjiG_pVvHcDIjvmaCd_qr/exec';

function lcbdGetDeviceId() {
    var k = 'lcbd_did';
    var id = localStorage.getItem(k);
    if (!id) {
        id = 'D' + Date.now().toString(36) + Math.random().toString(36).substr(2, 8);
        localStorage.setItem(k, id);
    }
    return id;
}

function lcbdShowGate() {
    document.getElementById('lcbdGate').style.display = 'block';
    document.getElementById('lcbdApp').style.display = 'none';
}

function lcbdShowApp() {
    document.getElementById('lcbdGate').style.display = 'none';
    document.getElementById('lcbdApp').style.display = 'block';
    lcbdCheckMicPerm();
}

function lcbdCheckMicPerm() {
    if (localStorage.getItem('lcbd_mic_ok') === '1') return;
    var banner = document.getElementById('lcbdMicBanner');
    if (!banner) return;
    if (navigator.permissions) {
        navigator.permissions.query({name:'microphone'}).then(function(r) {
            if (r.state === 'granted') {
                localStorage.setItem('lcbd_mic_ok', '1');
            } else {
                banner.style.display = 'flex';
            }
        }).catch(function() {
            banner.style.display = 'flex';
        });
    } else {
        banner.style.display = 'flex';
    }
}

window.lcbdRequestMic = function() {
    var banner = document.getElementById('lcbdMicBanner');
    navigator.mediaDevices.getUserMedia({audio:true})
    .then(function(stream) {
        stream.getTracks().forEach(function(t){ t.stop(); });
        localStorage.setItem('lcbd_mic_ok', '1');
        if (banner) banner.style.display = 'none';
    })
    .catch(function() {
        if (banner) {
            banner.innerHTML = '<span style="font-size:22px">&#10060;</span><div style="flex:1;font-size:13px;color:#92400e;font-weight:600">Permission denied &#8212; Settings &#8594; Apps &#8594; Learncarebd &#8594; Microphone &#8594; Allow</div>';
            banner.style.display = 'flex';
        }
    });
};

    var code = document.getElementById('lcbdCodeInput').value.trim().toUpperCase();

    var cbName = 'lcbdCb' + Date.now();

    window[cbName] = function(res) {

// Format code input automatically (LCBD-XXXX-XXXX)
function lcbdStartApp() {
    var inp = document.getElementById('lcbdCodeInput');
    if (inp) {
        inp.addEventListener('input', function() {
            var v = this.value.replace(/[^A-Z0-9]/gi, '').toUpperCase();
            var parts = [];
            if (v.length > 0) parts.push(v.substr(0, 4));
            if (v.length > 4) parts.push(v.substr(4, 4));
            if (v.length > 8) parts.push(v.substr(8, 4));
            this.value = parts.join('-');
        });
        inp.addEventListener('keydown', function(e) {
            if (e.key === 'Enter') lcbdActivate();
        });
    }

    // Check if already activated
    if (localStorage.getItem('lcbd_activated') === '1') {
        lcbdShowApp();
        lcbdInit();
    } else {
        lcbdShowGate();
    }
}

} else {
    lcbdStartApp();
}

function lcbdScore(h,e){
    if(!h) return 0;
    if(h===e) return 100;
    if(h.includes(e)) return 95;
    const d=lcbdLev(h,e);
    return Math.max(0,Math.round((1-d/Math.max(h.length,e.length))*100));
}

function lcbdLev(a,b){
    const dp=Array.from({length:b.length+1},(_,i)=>[i]);
    for(let j=0;j<=a.length;j++) dp[0][j]=j;
    for(let i=1;i<=b.length;i++)
        for(let j=1;j<=a.length;j++)
            dp[i][j]=b[i-1]===a[j-1]?dp[i-1][j-1]:1+Math.min(dp[i-1][j-1],dp[i][j-1],dp[i-1][j]);
    return dp[b.length][a.length];
}
